(function(){
 var root=document.getElementById('gta-tz-page');if(!root)return;
 var cards=[].slice.call(root.querySelectorAll('.gta-tz-card'));

 var none=document.getElementById('gta-tz-none');
 var formatters={};
 function midnightUTC(zone){
   // Find the UTC instant corresponding to local 2026-11-19 00:00 in the IANA time zone.
   // Fixed-point adjustment avoids manually hard-coding DST offsets.
   var wanted=Date.UTC(2026,10,19,0,0,0);
   var guess=wanted;
   var f=formatters[zone]||(formatters[zone]=new Intl.DateTimeFormat('en-US',{timeZone:zone,year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit',second:'2-digit',hourCycle:'h23'}));
   for(var i=0;i<4;i++){
     var x={};f.formatToParts(new Date(guess)).forEach(function(p){if(p.type!=='literal')x[p.type]=Number(p.value);});
     var local=Date.UTC(x.year,x.month-1,x.day,x.hour,x.minute,x.second);
     guess+=wanted-local;
   }
   return guess;
 }
 cards.forEach(function(card){try{card._target=midnightUTC(card.getAttribute('data-zone'));}catch(e){card._target=NaN;}});
 function render(){
   var now=Date.now();
   cards.forEach(function(card){
     if(!Number.isFinite(card._target))return;
     var secs=Math.max(0,Math.floor((card._target-now)/1000));
     var nums={days:Math.floor(secs/86400),hours:Math.floor(secs%86400/3600),minutes:Math.floor(secs%3600/60),seconds:secs%60};
     Object.keys(nums).forEach(function(key){var el=card.querySelector('[data-unit="'+key+'"]');if(el)el.textContent=String(nums[key]).padStart(2,'0');});
     if(secs===0){var status=card.querySelector('.gta-tz-status');if(status)status.textContent='Illustrative target time reached';}
   });
   var main=document.getElementById('gta-tz-main');
   if(main){var t=Math.max(0,Math.floor((Date.UTC(2026,10,19,5,0,0)-now)/1000));main.textContent=Math.floor(t/86400)+' DAYS · '+String(Math.floor(t%86400/3600)).padStart(2,'0')+' HOURS · '+String(Math.floor(t%3600/60)).padStart(2,'0')+' MINUTES · '+String(t%60).padStart(2,'0')+' SECONDS';}
 }

 render();setInterval(render,1000);
})();
